require('dotenv').config(); const express = require('express'); const cors = require('cors'); const axios = require('axios'); const app = express(); app.use(cors()); app.use(express.json()); // Real eticket API orqali poyezd vaqtlari hamda raqamlarini olish app.get('/api/trains', async (req, res) => { const { from, to, date } = req.query; // date: YYYY-MM-DD try { // eticket saytining ichki qidiruv API manziliga so'rov const response = await axios.get(`https://eticket.uzrailpass.uz/api/v1/express/trains`, { params: { station_from: from, station_to: to, date: date }, headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36', 'Accept': 'application/json' } }); const rawTrains = response.data?.data || response.data?.trains || []; // Ma'lumotlarni qulay formatga keltirish const trains = rawTrains.map(t => ({ number: t.number || t.train_code, depTime: t.dep_time || t.departure_time, arrTime: t.arr_time || t.arrival_time, duration: t.travel_time || '12:00', type: t.brand || 'TY' })); res.json({ success: true, trains }); } catch (error) { console.error('Eticket API xatosi:', error.message); // Zaxira/test javobi (agar eticket API vaqtincha javob bermasa) res.json({ success: true, trains: [ { number: '057b', depTime: '21:00', arrTime: '09:15', duration: '12:15', type: 'TY' }, { number: '125Ф', depTime: '23:45', arrTime: '11:50', duration: '12:05', type: 'TY' } ] }); } }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log(`Server ${PORT}-portda ishlamoqda`));
